\begin{frame}{Why this problem}
\begin{columns}[T]
\begin{column}{0.42\textwidth}
\begin{itemize}\setlength{\itemsep}{3pt}
\item Chat models read BPE tokens; a tokenizer-free model reads raw bytes
\item Bytes: no unknown words, robust to typos, code, other scripts
\item Cost: about 4 times more positions, so a 4 times larger KV cache
\item The KV cache, not the weights, limits batch size and context length
\item BLT already computes next-byte entropy to place patches: free signal?
\end{itemize}
\end{column}
\begin{column}{0.58\textwidth}\fig{presentation/mermaid/mermaid_01.png}\end{column}
\end{columns}
\end{frame}

\begin{frame}{How it relates to the models people use}
\fig[0.98]{figures/fig_public.pdf}
\scriptsize Llama 3 8B: 32 layers, 8 KV heads (grouped-query attention to shrink the cache), 128k tokens; Llama 2 7B: 32k BPE vocabulary, 4k context. Ours: 13.3M parameters, trained from scratch on one GPU. We compare cost structure and methods, not chat quality.
\end{frame}

\begin{frame}{The task and the metric}
\begin{itemize}\setlength{\itemsep}{4pt}
\item Task: next-symbol prediction (language modelling), scored in \textbf{bits per byte (\bpb)}
\item \bpb{} is tokenizer-independent: 4-gram, BPE and byte models on identical documents
\item Datasets: TinyStories (easy), WikiText-103 (hard, long), The Stack Smol (code mix), PG-19 (books)
\item Baselines: Kneser-Ney byte 4-gram (classical), BPE Transformer 13.8M (modern)
\item Evaluation: three seeds, paired bootstrap intervals, WikiText test split used once
\end{itemize}
\end{frame}

\begin{frame}{Main test: is entropy a good keep signal?}
\fig[0.52]{figures/fig_scoreboard.pdf}
\begin{itemize}\setlength{\itemsep}{2pt}\small
\item Worse than random in 68 and than recency in 70 of 72 comparisons, better in none
\item Holds in every seed, fixed-stride models, code mix, 16k context, and the test split (36 of 36)
\item SnapKV-style (online): better than recency in 24 of 24
\end{itemize}
\end{frame}

\begin{frame}{More training makes it worse, not better}
\fig[0.98]{figures/fig_scale.pdf}
\scriptsize Penalty against random at 64 patches: +0.002 (3,000 steps) to +0.024 to +0.029 (30,000 steps). SnapKV-style gain grows several times as well.
\end{frame}

\begin{frame}{Why: what the entropy rule keeps}
\fig[0.98]{figures/fig_kept.pdf}
\begin{itemize}\setlength{\itemsep}{2pt}\small
\item It keeps one- and two-byte fragments (98.0\% and 99.7\% of kept patches) and drops patches with words
\item Old high-entropy patches also receive less attention in middle and upper layers
\end{itemize}
\end{frame}

\begin{frame}{Where the extra loss falls}
\fig[0.85]{figures/fig_terciles.pdf}
\scriptsize Keeping high-entropy patches does not protect the high-entropy bytes and also costs the easy ones. The top 1\% of bytes carry 18 to 30\% of the extra loss.
\end{frame}

\begin{frame}{What works: eviction, quantization, SnapKV-style}
\fig[0.98]{figures/fig_frontier.pdf}
\scriptsize WikiText: 128 patches in 4-bit keep 7.4\% of the cache bytes for +0.0012 \bpb{} (SnapKV-style); keeping high entropy never reaches the frontier.
\end{frame}

\begin{frame}{Bounded-memory streaming}
\fig[0.98]{figures/fig_stream.pdf}
\scriptsize A fixed recency cache beats resetting the context in 3 of 3 seeds and on 20 PG-19 books (by 0.0086 \bpb).
\end{frame}

\begin{frame}{Error analysis: generations diverge}
\footnotesize TinyStories, prompt of 700 bytes, 32 patches (real output, greedy):
\begin{itemize}\setlength{\itemsep}{3pt}
\item full: \texttt{find your friends."  Roxy was so happy to help Roxy.}
\item recency (diverges at byte 17): \texttt{find your friends to climb the leaves."...}
\item keep high entropy (byte 25): \texttt{...Roxyy said, "Thank you, Billy.}
\end{itemize}
Per-byte: \texttt{They argue[d]} costs 0.02 bits with the full cache and 5.97 under entropy eviction. Small models repeat; these are anecdotes of where the compressed cache changes the prediction.
\end{frame}

\begin{frame}{Challenges and how we handled them}
\small
\begin{itemize}\setlength{\itemsep}{3pt}
\item Seed variance as large as gaps: three seeds, paired bootstrap; a 3.3\% claim retired
\item Byte-share accounting error found by cross-table check: fixed, all tables regenerated
\item Under-trained 16k models gave a wrong conclusion: tuned, penalty does not fade
\item Offline attention scores approximate: online implementation built and tested
\item Silent OOM corrupted a cache: caught from logs, queue hardened
\item Gated code data, PG-19 loader, cluster limits: workarounds recorded in \texttt{CHALLENGES.md}
\end{itemize}
\end{frame}

\begin{frame}{Scope, limits and what is next}
\begin{itemize}\setlength{\itemsep}{3pt}
\item 13.3M parameters; the models do not retrieve (planted-name test): retrieval tasks untested
\item Cache sizes computed except recency; FLOPs analytic; PG-19 confounds entropy with distance
\item Next: larger models, positions renumbered inside the cache, attention policies in step decoding
\end{itemize}
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}\setlength{\itemsep}{4pt}
\item \textbf{New:} controlled test of entropy-ranked cache retention on a byte-level patch model; online attention policies over patch caches; measured bounded-memory streaming
\item \textbf{Improved:} 7.4\% of the cache for +0.0012 \bpb; SnapKV-style beats recency 24/24; constant memory and time; 2.8x fewer FLOPs than a byte Transformer
\item \textbf{Did not work:} entropy as the keep signal (68 to 70 of 72 worse), and it grows worse with training
\item \textbf{Open:} larger models and retrieval
\end{itemize}
